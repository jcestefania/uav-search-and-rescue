# Calcula el perímetro del área donde volará el dron
def area(centroide: tuple, radio: int, tipo: str, escala: float, num_puntos=36):
    longitud_centro, latitud_centro = centroide

    radio_grados_lat = (radio*escala) / 111000  # Convertir el radio a grados de latitud
    radio_grados_lon = (radio*escala) / (111000 * math.cos(math.radians(latitud_centro)))  # Convertir el radio a grados de longitud

    coordenadas_perimetro = []

    if tipo == "circular":
        # Crear puntos en un círculo alrededor del centroide
        for i in range(num_puntos):
            angulo = 2 * math.pi * i / num_puntos
            delta_lat = radio_grados_lat * math.sin(angulo)
            delta_lon = radio_grados_lon * math.cos(angulo)
            lat = latitud_centro + delta_lat
            lon = longitud_centro + delta_lon
            coordenadas_perimetro.append((lon, lat))

    elif tipo == "rectangular":
        # Crear los vértices de un rectángulo alrededor del centroide
        coordenadas_perimetro = [
            (longitud_centro - radio_grados_lon, latitud_centro - radio_grados_lat),  # Suroeste
            (longitud_centro + radio_grados_lon, latitud_centro - radio_grados_lat),  # Sureste
            (longitud_centro + radio_grados_lon, latitud_centro + radio_grados_lat),  # Noreste
            (longitud_centro - radio_grados_lon, latitud_centro + radio_grados_lat),  # Noroeste
            (longitud_centro - radio_grados_lon, latitud_centro - radio_grados_lat)   # Cerrar el rectángulo
        ]

    else:
        raise ValueError("Tipo de área no válido. Usa 'circular' o 'rectangular'.")

    return coordenadas_perimetro


def numero_cuadriculas(radio, escala, tam_cuadricula):
    # Asumiendo que el área siempre va a ser un cuadrado (está en metros)

    cuadriculas_ancho = (2*radio*escala) / tam_cuadricula
    cuadriculas_largo = (2*radio*escala) / tam_cuadricula

    #total_cuadriculas = cuadriculas_ancho * cuadriculas_largo

    return cuadriculas_ancho, cuadriculas_largo # Aunque sea un cuadrado, paso ambos por si en un futuro cambio la forma del perímetro

def delta_lat_lon(top_left_lat, tam_cuadricula):
    """Calcular cuántos grados en latitud y longitud equivale 1 metro (diferencial)
    Regla de 3: calculo cuánto equivale 1 metro en º de latitud y longitud y luego multiplico el tamaño de la cuadricula por eso"""

    radio_tierra = 6378.137 # en km
    pi = math.pi

    m2degree = (1 / ((2 * pi /360) * radio_tierra)) / 1000 # A cuántos grados en latitud (también se usa para la longitud) corresponde un metro
    delta_lat = (tam_cuadricula) * m2degree

    delta_lon = (tam_cuadricula) * m2degree / math.cos(math.radians(top_left_lat))

    return delta_lat, delta_lon

def coords_cuadricula(coordenadas_perimetro, cuadriculas_ancho, cuadriculas_largo, delta_lat, delta_lon):
    
    suroeste, sureste, noreste, noroeste, _ = coordenadas_perimetro

    coords_malla = []

    for i in range(int(cuadriculas_ancho)):
        for j in range(int(cuadriculas_largo)):

            # Calcular las coordenadas de los vértices del cuadrado
            lon_min = noroeste[0] + (i * delta_lon)  # Límite oeste del cuadrado
            lon_max = lon_min + delta_lon            # Límite este del cuadrado
            lat_max = noroeste[1] - (j * delta_lat)  # Límite norte del cuadrado
            lat_min = lat_max - delta_lat            # Límite sur del cuadrado

  
            cuadrado = [
                [lon_min, lat_max],  # Esquina noroeste
                [lon_max, lat_max],  # Esquina noreste
                [lon_max, lat_min],  # Esquina sureste
                [lon_min, lat_min],  # Esquina suroeste
                [lon_min, lat_max]   # Repetir la esquina noroeste para cerrar el cuadrado
            ]

            coords_malla.append(cuadrado)

    return coords_malla
